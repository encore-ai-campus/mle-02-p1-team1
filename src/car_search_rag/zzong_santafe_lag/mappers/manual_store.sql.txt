-- [프로젝트 추가] 개인 테이블 전용 저장·조회 SQL Mapper입니다.
-- :이름은 파이썬에서 전달할 값입니다. 문자열을 SQL에 직접 이어 붙이지 않습니다.
-- 생성 SQL은 별도 db 폴더에 있어 mapper 등록만으로 테이블을 만들지 않습니다.

-- name: list_tables
SELECT table_name FROM information_schema.tables
WHERE table_schema = 'zzong_santafe_lag' AND table_type = 'BASE TABLE'
ORDER BY table_name;

-- name: connection_info
-- [프로젝트 추가] 주소·비밀번호 없이 연결한 DB와 읽기 전용 상태를 표시합니다.
SELECT current_database() AS database_name, current_user AS database_user,
       current_setting('transaction_read_only') AS read_only,
       e.extversion AS vector_version, n.nspname AS vector_schema
FROM pg_extension e JOIN pg_namespace n ON n.oid = e.extnamespace
WHERE e.extname = 'vector';

-- name: personal_table_counts
-- 개인 여섯 테이블만 읽습니다. 공유 public 테이블의 자료는 조회하지 않습니다.
SELECT 'documents' AS table_name, count(*) AS row_count FROM zzong_santafe_lag.documents
UNION ALL SELECT 'processing_runs', count(*) FROM zzong_santafe_lag.processing_runs
UNION ALL SELECT 'parent_records', count(*) FROM zzong_santafe_lag.parent_records
UNION ALL SELECT 'chunks', count(*) FROM zzong_santafe_lag.chunks
UNION ALL SELECT 'images', count(*) FROM zzong_santafe_lag.images
UNION ALL SELECT 'record_images', count(*) FROM zzong_santafe_lag.record_images;

-- name: personal_table_sizes
-- 테이블·색인·TOAST 크기이며 Storage 파일과 DB 전체 관리 영역은 제외합니다.
SELECT c.relname AS table_name, pg_total_relation_size(c.oid) AS bytes
FROM pg_class c JOIN pg_namespace n ON n.oid = c.relnamespace
WHERE n.nspname = 'zzong_santafe_lag' AND c.relkind = 'r'
ORDER BY c.relname;

-- name: search_chunks
-- <=>는 코사인 거리입니다. 현재 TF-IDF·질문 목적 결합 검색의 완성 대체 기능은 아닙니다.
-- 같은 처리 버전의 완료 자료만 검색하고, 검토 상태는 부모 컬럼에서 읽습니다.
SELECT c.record_id AS chunk_record_id, c.content AS chunk_content,
       p.record_id AS parent_record_id, p.title, p.pdf_page_number,
       p.manual_page_number, p.source_pages, p.verification_status,
       1 - (c.embedding <=> :query_embedding) AS similarity
FROM zzong_santafe_lag.chunks c
JOIN zzong_santafe_lag.parent_records p ON p.id = c.parent_id
JOIN zzong_santafe_lag.processing_runs r ON r.id = c.run_id
WHERE c.run_id = :run_id AND r.status = 'ready'
ORDER BY c.embedding <=> :query_embedding, c.record_id
LIMIT :limit;

-- name: get_run_search_counts
-- [프로젝트 추가] 전체 PDF가 아닌 현재 run_id의 자료 개수만 확인합니다.
SELECT
    (SELECT count(*) FROM zzong_santafe_lag.parent_records WHERE run_id = :run_id) AS parent_count,
    (SELECT count(*) FROM zzong_santafe_lag.chunks WHERE run_id = :run_id) AS chunk_count;

-- name: search_parent_chunks
-- [프로젝트 추가] 모든 청크의 거리를 계산하고 부모마다 가장 가까운 청크 한 개를 남깁니다.
-- DISTINCT ON은 괄호 안의 부모 번호가 같은 행 중 정렬 첫 행을 선택합니다.
-- 부모별 대표를 다시 유사도 순으로 정렬합니다. 표본·다른 작업 벡터는 포함하지 않습니다.
WITH representatives AS (
    SELECT DISTINCT ON (p.id)
           c.record_id AS chunk_record_id, c.content AS chunk_content,
           p.record_id AS parent_record_id, p.title, p.pdf_page_number,
           p.manual_page_number, p.source_pages, p.verification_status,
           1 - (c.embedding <=> :query_embedding) AS similarity
    FROM zzong_santafe_lag.chunks c
    JOIN zzong_santafe_lag.parent_records p ON p.id = c.parent_id
    JOIN zzong_santafe_lag.processing_runs r ON r.id = c.run_id
    WHERE c.run_id = :run_id AND r.status = 'ready'
    ORDER BY p.id, c.embedding <=> :query_embedding, c.record_id
)
SELECT * FROM representatives
ORDER BY similarity DESC, chunk_record_id
LIMIT :limit;

-- [프로젝트 추가] 원본 다운로드 확인 뒤 보관 경로 세 항목만 변경합니다. 설명·원문·벡터는 그대로입니다.
