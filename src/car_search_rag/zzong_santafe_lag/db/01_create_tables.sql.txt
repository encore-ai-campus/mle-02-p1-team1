-- [프로젝트 추가] 개인 싼타페 자료용 6개 테이블을 만드는 SQL입니다.
-- DBeaver의 zzong 연결에서 검토 후 승인받아 전체 스크립트를 실행합니다.
-- 현재 모델: jhgan/ko-sroberta-multitask-mrl / 768차원 / 최대 128토큰.
-- 기존 팀 테이블·Storage 시스템 테이블·공통 확장은 수정하지 않습니다.
-- 같은 스키마가 이미 있으면 중단하며 덮어쓰지 않습니다.

BEGIN;

-- 실제 pgvector 설치 위치를 찾습니다. 없으면 공통 확장을 만들지 않고 중단합니다.
DO $$
DECLARE vector_schema text;
BEGIN
    IF EXISTS (SELECT 1 FROM pg_namespace WHERE nspname = 'zzong_santafe_lag') THEN
        RAISE EXCEPTION '개인 스키마가 이미 있습니다. 기존 구조부터 확인하세요.';
    END IF;
    SELECT n.nspname INTO vector_schema
    FROM pg_extension e JOIN pg_namespace n ON n.oid = e.extnamespace
    WHERE e.extname = 'vector';
    IF vector_schema IS NULL THEN
        RAISE EXCEPTION 'pgvector가 없습니다. Supabase의 vector 확장 상태부터 확인하세요.';
    END IF;
    -- 자료형 검색 경로는 이번 트랜잭션에만 적용합니다.
    PERFORM set_config('search_path', format('pg_catalog,%I', vector_schema), true);
END $$;

-- 스키마는 개인 테이블을 모아 놓는 공간입니다.
CREATE SCHEMA zzong_santafe_lag;
COMMENT ON SCHEMA zzong_santafe_lag IS 'zzong 개인 싼타페 PDF 챗봇';

-- 1. 같은 PDF를 여러 번 처리해도 파일 정보는 한 번만 보관합니다.
CREATE TABLE zzong_santafe_lag.documents (
    id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    file_name text NOT NULL,
    file_sha256 text NOT NULL UNIQUE CHECK (file_sha256 ~ '^[0-9a-f]{64}$'),
    total_pages integer NOT NULL CHECK (total_pages > 0),
    local_path text,
    storage_bucket text,
    storage_path text,
    created_at timestamptz NOT NULL DEFAULT now(),
    CHECK ((storage_bucket IS NULL) = (storage_path IS NULL))
);
COMMENT ON TABLE zzong_santafe_lag.documents IS 'PDF 파일 정보와 보관 경로';
COMMENT ON COLUMN zzong_santafe_lag.documents.file_sha256 IS '파일 내용의 SHA-256 식별값; 같은 파일 중복 등록 방지';

-- 2. 모델·청킹 설정을 바꾼 결과는 새 처리 작업으로 구분합니다.
CREATE TABLE zzong_santafe_lag.processing_runs (
    id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    document_id uuid NOT NULL REFERENCES zzong_santafe_lag.documents(id),
    pipeline_version text NOT NULL,
    model_name text NOT NULL,
    model_revision text NOT NULL,
    embedding_dimension integer NOT NULL CHECK (embedding_dimension = 768),
    token_budget integer NOT NULL CHECK (token_budget BETWEEN 1 AND 128),
    normalized boolean NOT NULL CHECK (normalized),
    settings jsonb NOT NULL DEFAULT '{}'::jsonb CHECK (jsonb_typeof(settings) = 'object'),
    status text NOT NULL DEFAULT 'building' CHECK (status IN ('building', 'ready')),
    created_at timestamptz NOT NULL DEFAULT now(),
    UNIQUE (id, document_id)
);
COMMENT ON TABLE zzong_santafe_lag.processing_runs IS '청킹·모델 설정과 처리 버전; ready만 검색 대상';
COMMENT ON COLUMN zzong_santafe_lag.processing_runs.model_revision IS '실제로 사용한 로컬 모델 스냅샷 버전';
COMMENT ON COLUMN zzong_santafe_lag.processing_runs.settings IS '코드 버전·배치 크기·CPU 스레드 등 실험 설정';

-- 3. 주제 전체 글과 출처를 보관합니다. 검색 글을 고쳐도 원문은 남습니다.
CREATE TABLE zzong_santafe_lag.parent_records (
    id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    run_id uuid NOT NULL,
    document_id uuid NOT NULL,
    record_id text NOT NULL,
    title text NOT NULL,
    raw_text text NOT NULL,
    content text NOT NULL CHECK (length(btrim(content)) > 0),
    pdf_page_number integer NOT NULL CHECK (pdf_page_number > 0),
    manual_page_number integer CHECK (manual_page_number > 0),
    source_pages integer[] NOT NULL CHECK (cardinality(source_pages) > 0 AND 0 < ALL(source_pages)),
    content_type text NOT NULL,
    verification_status text NOT NULL CHECK (verification_status IN
        ('sample_verified', 'visually_reviewed_source', 'auto_draft_needs_review')),
    review_flags jsonb NOT NULL DEFAULT '[]'::jsonb CHECK (jsonb_typeof(review_flags) = 'array'),
    metadata jsonb NOT NULL CHECK (jsonb_typeof(metadata) = 'object'),
    created_at timestamptz NOT NULL DEFAULT now(),
    FOREIGN KEY (run_id, document_id) REFERENCES zzong_santafe_lag.processing_runs(id, document_id),
    UNIQUE (run_id, record_id),
    UNIQUE (id, run_id),
    UNIQUE (id, document_id)
);
COMMENT ON TABLE zzong_santafe_lag.parent_records IS '부모 기록: 주제·원문·검색 글·출처·검토 상태';
COMMENT ON COLUMN zzong_santafe_lag.parent_records.raw_text IS '수정하지 않고 보관하는 추출 원문';
COMMENT ON COLUMN zzong_santafe_lag.parent_records.content IS '검색용 글; 확인한 그림 설명 포함';
COMMENT ON COLUMN zzong_santafe_lag.parent_records.metadata IS '기존 출처 딕셔너리 전체; 표 행·각주·필수 문맥 연결·원문 구간 보존';
COMMENT ON COLUMN zzong_santafe_lag.parent_records.record_id IS '기존 파이썬 기록 ID; 같은 처리 작업 안에서만 고유';

-- 4. 실제 모델에 넣은 글과 해당 벡터를 함께 저장합니다. 그림 자체의 벡터는 아닙니다.
CREATE TABLE zzong_santafe_lag.chunks (
    id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    run_id uuid NOT NULL REFERENCES zzong_santafe_lag.processing_runs(id),
    parent_id uuid NOT NULL,
    record_id text NOT NULL,
    chunk_index integer NOT NULL CHECK (chunk_index >= 0),
    content text NOT NULL CHECK (length(btrim(content)) > 0),
    content_sha256 text NOT NULL CHECK (content_sha256 ~ '^[0-9a-f]{64}$'),
    token_count integer NOT NULL CHECK (token_count BETWEEN 1 AND 128),
    parent_content_start integer NOT NULL CHECK (parent_content_start >= 0),
    parent_content_end integer NOT NULL CHECK (parent_content_end > parent_content_start),
    embedding vector(768) NOT NULL,
    metadata jsonb NOT NULL CHECK (jsonb_typeof(metadata) = 'object'),
    created_at timestamptz NOT NULL DEFAULT now(),
    FOREIGN KEY (parent_id, run_id) REFERENCES zzong_santafe_lag.parent_records(id, run_id),
    UNIQUE (run_id, record_id),
    UNIQUE (parent_id, chunk_index)
);
COMMENT ON TABLE zzong_santafe_lag.chunks IS '제목 포함 실제 모델 입력 글과 768차원 임베딩';
COMMENT ON COLUMN zzong_santafe_lag.chunks.content_sha256 IS '글 변경 확인값; 변경한 글은 재임베딩 필요';
COMMENT ON COLUMN zzong_santafe_lag.chunks.parent_content_start IS '부모 content의 시작 위치; raw_text 위치 아님';
COMMENT ON COLUMN zzong_santafe_lag.chunks.parent_content_end IS '부모 content의 끝 위치; 파이썬 슬라이스처럼 끝 제외';
COMMENT ON COLUMN zzong_santafe_lag.chunks.metadata IS '기존 청크 정보 보존; 검토 상태의 조회 기준은 부모 컬럼';

-- 5. 아직 파일을 꺼내지 않은 그림은 이름·크기·파일 경로가 NULL일 수 있습니다.
-- I1.jpg처럼 반복되는 이름 대신 PDF·쪽수·내부 이미지 키로 구분합니다.
CREATE TABLE zzong_santafe_lag.images (
    id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    document_id uuid NOT NULL REFERENCES zzong_santafe_lag.documents(id),
    pdf_page_number integer NOT NULL CHECK (pdf_page_number > 0),
    pdf_image_key jsonb NOT NULL CHECK (pdf_image_key <> 'null'::jsonb),
    image_key_sha256 text NOT NULL CHECK (image_key_sha256 ~ '^[0-9a-f]{64}$'),
    file_name text,
    width integer CHECK (width > 0),
    height integer CHECK (height > 0),
    local_path text,
    storage_bucket text,
    storage_path text,
    upload_status text NOT NULL DEFAULT 'reference_only'
        CHECK (upload_status IN ('reference_only', 'local_available', 'uploaded')),
    created_at timestamptz NOT NULL DEFAULT now(),
    CHECK ((width IS NULL) = (height IS NULL)),
    CHECK ((storage_bucket IS NULL) = (storage_path IS NULL)),
    CHECK (upload_status <> 'local_available' OR local_path IS NOT NULL),
    CHECK (upload_status <> 'uploaded' OR storage_path IS NOT NULL),
    UNIQUE (document_id, pdf_page_number, image_key_sha256),
    UNIQUE (id, document_id)
);
COMMENT ON TABLE zzong_santafe_lag.images IS '그림 참조와 경로; 실제 파일은 Storage 또는 로컬에 보관';
COMMENT ON COLUMN zzong_santafe_lag.images.pdf_image_key IS 'pypdf 실제 내부 키; 중첩 키는 JSON 배열로 보존';
COMMENT ON COLUMN zzong_santafe_lag.images.image_key_sha256 IS '일정한 JSON 형식으로 바꾼 내부 키의 SHA-256 식별값';

-- 6. 한 그림을 여러 주제에 연결합니다. 설명과 검토 상태는 연결마다 보관합니다.
CREATE TABLE zzong_santafe_lag.record_images (
    document_id uuid NOT NULL,
    parent_id uuid NOT NULL,
    image_id uuid NOT NULL,
    description text,
    linkage_status text NOT NULL DEFAULT 'unreviewed_page_reference'
        CHECK (linkage_status IN ('unreviewed_page_reference', 'sample_linked', 'visually_verified')),
    metadata jsonb NOT NULL DEFAULT '{}'::jsonb CHECK (jsonb_typeof(metadata) = 'object'),
    PRIMARY KEY (parent_id, image_id),
    FOREIGN KEY (parent_id, document_id) REFERENCES zzong_santafe_lag.parent_records(id, document_id),
    FOREIGN KEY (image_id, document_id) REFERENCES zzong_santafe_lag.images(id, document_id)
);
COMMENT ON TABLE zzong_santafe_lag.record_images IS '같은 PDF의 주제·그림 연결; 연결 검토와 업로드 상태는 별개';

-- 관계 조회용 기본 색인입니다. 벡터 근사 검색 색인은 이후 비교해서 결정합니다.
CREATE INDEX parent_records_run_type_idx ON zzong_santafe_lag.parent_records(run_id, content_type);
CREATE INDEX chunks_run_idx ON zzong_santafe_lag.chunks(run_id);
CREATE INDEX record_images_image_idx ON zzong_santafe_lag.record_images(image_id);

-- API 공개 정책은 만들지 않습니다. DBeaver/SQL Mapper 역할 권한은 별도로 확인합니다.
-- 소유자·관리자는 RLS를 우회할 수 있어 개인 이름만으로 접근 차단을 보장하지 않습니다.
ALTER TABLE zzong_santafe_lag.documents ENABLE ROW LEVEL SECURITY;
ALTER TABLE zzong_santafe_lag.processing_runs ENABLE ROW LEVEL SECURITY;
ALTER TABLE zzong_santafe_lag.parent_records ENABLE ROW LEVEL SECURITY;
ALTER TABLE zzong_santafe_lag.chunks ENABLE ROW LEVEL SECURITY;
ALTER TABLE zzong_santafe_lag.images ENABLE ROW LEVEL SECURITY;
ALTER TABLE zzong_santafe_lag.record_images ENABLE ROW LEVEL SECURITY;

-- 성공한 경우에만 빈 테이블 구조를 확정합니다. 자료 저장·Storage 생성은 별도 단계입니다.
COMMIT;
